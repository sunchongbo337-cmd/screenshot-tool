// ① 文件名预览 — packages/editor-react/src/settings/EditorSettingsDialog.tsx
export function previewSaveFilename(pattern: string, seqNumber = 1): string {
  return pattern
    .replace(/\{timestamp\}/g, String(Date.now()))
    .replace(/\{date\}/g, date).replace(/\{time\}/g, time)
    .replace(/\{n\}/g, String(seq)).replace(/\{seq\}/g, String(seq)) + '.png';
}

// packages/web-plugin/src/index.ts
export function createEditor(params: CreateEditorParams): EditorInstance {
  const root = createRoot(params.container);
  const handleRef = React.createRef<EditorWidgetHandle>();
  root.render(React.createElement(EditorWidget, { ref: handleRef, image, options }));
  return {
    setTool: (t) => handleRef.current?.setTool(t),
    selectMosaicsSameRow: () => handleRef.current!.selectMosaicsSameRow(),
    export: (o) => handleRef.current!.export(o),
    destroy: () => { handleRef.current?.destroy(); root.unmount(); }
  };
}

// 同行算法 — EditorWidget.tsx
function collectSameRowMosaicIds(doc, seedId): string[] {
  const midY = b.y + b.height / 2;  // 种子水平中线
  for (const n of doc.nodes) {
    if (midY >= r.y && midY <= r.y + r.height) out.push(n.id);
  }
}

// 同列算法
function collectSameColumnMosaicIds(doc, seedId): string[] {
  const midX = b.x + b.width / 2;   // 种子垂直中线
  if (midX >= r.x && midX <= r.x + r.width) out.push(n.id);
}

// 命名解析（支持 {n} 自动递增）
export function resolveTemplateName(pattern: string, existing: string[]): string {
  const match = pattern.match(/(.+)\{n\}$/);
  if (match) {
    const prefix = match[1];
    const nums = existing
      .map(n => n.slice(prefix.length))
      .filter(n => /^\d+$/.test(n))
      .map(Number);
    let next = 1;
    while (nums.includes(next)) next++;
    return `${prefix}${next}`;
  }
  return pattern;
}
